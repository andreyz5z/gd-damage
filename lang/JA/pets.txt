# 「アクティブペット」ウィンドウと、他のウィンドウにあるペットの行。
# ペットとそのスキルの名前はゲームから取っています。

pets.title=アクティブペット — {0}
pets.checkbox={0}   ({1}, Lv. {2})
pets.limit=同時に最大 {0}
pets.lifetime=存続 {0} 秒
pets.life=ヘルス: 基本 {0} × (1 + 自身 {1}% + こちら {2}%) = {3}
pets.lifeAura=ヘルス: 基本 {0} × (1 + 自身 {1}% + こちら {2}% + オーラ {3}%) = {4}
pets.armour=装甲: 基本 {0} × (1 + 自身 {1}% + こちら {2}%) = {3}   (ゲームはペットの装甲を表示しません — これは独自の値です)
pets.auras=オーラの影響: {0}
pets.none=このキャラクターにはペットを召喚するスキルがありません。
pets.note=ヘルス: 基本 × (1 + 自身% + ペット ボーナス% + オーラ%)。スキルのダメージにはペット ボーナスが掛かります — この規則はゲームの 7 つの数値で測定しました。

pets.level=Lv. {0}
pets.levelUnknown=ペットのレコードにレベルの記載がありません
pets.attack={0}   (ペット «{1}», {2})

# ペットの変換
pets.conversion.title=このペットの変換
pets.conversion.capped=1 つの種類から出ていくのは 100% までです: 100% は受け取り側の割合に応じて分けられます。括弧内はアイテムに実際に書かれている値です。
pets.conversion.line={0}% 変換: {1} → {2}
pets.conversion.recorded=(レコード上 {0}%)
